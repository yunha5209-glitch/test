구글 시트 실시간 연동 설정 (Apps Script)

학생들이 [구글 시트로 제출] 버튼을 눌렀을 때 데이터가 선생님의 스프레드시트로 전송되도록 웹앱 URL을 등록해주세요.
(※ 구글 시트 메뉴: 확장 프로그램 > Apps Script > 아래 코드를 복사하여 넣고 '웹앱'으로 배포한 뒤 URL을 입력하세요.) 

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rowData = [];
  rowData.push(new Date()); // 제출시간 (A열)
  rowData.push(e.parameter.school); // 학교 (B열)
  rowData.push(e.parameter.grade); // 학년반 (C열)
  rowData.push(e.parameter.number); // 학번 (D열)
  rowData.push(e.parameter.name); // 이름 (E열)
  rowData.push(e.parameter.scraps); // 수집노트수 (F열)
  rowData.push(e.parameter.quizScore); // 퀴즈점수 (G열)
  rowData.push(e.parameter.motivation); // 탐구동기 (H열)
  rowData.push(e.parameter.analysis); // 분석 (I열)
  rowData.push(e.parameter.conclusion); // 결론 (J열)
  
  sheet.appendRow(rowData);
  return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT);
}
